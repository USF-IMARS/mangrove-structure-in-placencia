Sector Explorer

Trees were sampled across three lagoon sectors — upper, lower, and main creek — plus a handful of offshore caye sites not yet shown on this map. Click a sector below to see its structure: species mix, trunk diameter (DBH) and height distributions, and the underlying tree-level data. To see biomass by individual field station instead, visit Sampling Points.

// Sector polygon layer, built once. Hover state is tracked with a plain
// closure variable (not a reactive `mutable`) so hovering never re-triggers
// the layer rebuild — only clicks notify the rest of the page, via
// `mutable selectedSector`.
sectorGeoLayer = {
  let activeRegion = null;

  function baseStyle(region) {
    return {
      color: "#555",
      fillColor: sectorColors[region] ?? "#ccc",
      weight: 2,
      fillOpacity: 0.45,
    };
  }

  const layer = L.geoJSON(sectorsGeo, {
    style: (f) => baseStyle(f.properties.region),
    onEachFeature: (feature, lyr) => {
      const region = feature.properties.region;
      lyr.bindTooltip(sectorLabels[region] ?? region, { sticky: true, direction: "top" });
      lyr.on({
        mouseover: (e) => e.target.setStyle({ weight: 3, fillOpacity: 0.65 }),
        mouseout: (e) => {
          if (activeRegion !== region) e.target.setStyle(baseStyle(region));
        },
        click: (e) => {
          activeRegion = region;
          mutable selectedSector = region;
        },
      });
    },
  });
  layer.addTo(map);
  return layer;
}
Code 
// Reactive pass: re-applies the "selected" emphasis whenever selectedSector
// changes, without rebuilding the layer.
{
  sectorGeoLayer.eachLayer((lyr) => {
    const region = lyr.feature.properties.region;
    const selected = region === selectedSector;
    lyr.setStyle({
      weight: selected ? 4 : 2,
      fillOpacity: selected ? 0.75 : 0.45,
    });
  });
}
